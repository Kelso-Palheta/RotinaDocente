import {it,expect,vi,afterEach} from 'vitest';
const m=vi.hoisted(()=>({salvar:vi.fn(),atividade:{titulo:'Atividade sintética',professorId:'p',dataEntrega:'2099-01-01',turmaIds:['t'],bimestre:3,questoes:[],alunosPorTurma:{t:[{id:'a',nome:'Aluno fictício',dataNascimento:'1503',necessidades:['perfil-sintetico']}]}}}));
vi.mock('@/lib/firebase-admin',()=>({getAdminDb:()=>({doc:path=>({id:path.split('/').at(-1),get:async()=>path.startsWith('atividades/')&&!path.includes('/tokens/')?{exists:true,data:()=>m.atividade,id:'at'}:{exists:false,data:()=>null},set:m.salvar})})}));
import {GET} from '@/app/api/aluno/atividade/[id]/route';
import {POST} from '@/app/api/aluno/entregar/route';
import {encodeToken} from '@/utils/diario/tokenUtils';
afterEach(()=>m.salvar.mockClear());
it('REPRODUÇÃO: GET aceita identificador codificado sem documento de token emitido',async()=>{
 const token=encodeToken('aluno-nao-emitido','at');
 const res=await GET(new Request(`http://localhost/api/aluno/atividade/at?token=${token}`),{params:Promise.resolve({id:'at'})});
 expect(res.status).toBe(200);const data=await res.json();expect(data.alunoInfo).toBeNull();
 expect(data.atividade.alunosPorTurma.t[0].necessidades).toEqual(['perfil-sintetico']);
});
it('REPRODUÇÃO: POST aceita entrega sem sessão/token e turma/bimestre fornecidos pelo cliente',async()=>{
 const res=await POST(new Request('http://localhost/api/aluno/entregar',{method:'POST',headers:{'content-type':'application/json'},body:JSON.stringify({activityId:'at',alunoId:'a',turmaId:'outra',bimestre:4,respostas:{q:{resposta:'Resposta sintética'}}})}));
 expect(res.status).toBe(200);expect(m.salvar).toHaveBeenCalledWith(expect.objectContaining({turmaId:'outra',bimestre:4,status:'entregue'}),{merge:true});
});
